' LIVELLO 2: LOGICA DI BUSINESS
package "com.dietiestates25.service" <<Logic>> as service {
    class JwtService {
        +generateToken(UserDetails) : String
        +extractUsername(token) : String
    }

    interface StorageService <<interface>> {
        +uploadFile(file) : String
    }

    class LocalStorageService
    class S3StorageService
    
    StorageService <|.. LocalStorageService
    StorageService <|.. S3StorageService

    class AuthService {
        +register(RegisterRequest)
        +registerAgency(AgencyRegisterRequest)
        +login(LoginRequest) : AuthResponse
    }

    class PropertyService {
        +searchProperties(criteria, page, limit) : Page<PropertyDTO>
        +createProperty(req, agentEmail) : PropertyDTO
        +getProperty(id) : PropertyDTO
    }

    class AgentService {
        +createAgent(req)
        +getAgents() : List<AgentDTO>
        +getAgent(id) : AgentDTO
        +updateAgent(id, req)
    }

    class AgencyService {
        +getAgentsByAgency(agencyId) : List<AgentDTO>
        +getAgentsByManagerEmail(email) : List<AgentDTO>
    }

    class UserService {
        +getCurrentUserProfile(email) : UserDTO
        +updateProfile(req, email)
    }

    class ReviewService {
        +getReviewsByAgent(id) : List<ReviewDTO>
        +createReview(req, userEmail)
    }
}

' LIVELLO 3: PERSISTENZA
package "com.dietiestates25.repository" <<Data>> as repository {
    interface UserRepository <<interface>>
    interface AgentRepository <<interface>>
    interface AgencyRepository <<interface>>
    interface PropertyRepository <<interface>>
    interface AmenityRepository <<interface>>
    interface ReviewRepository <<interface>>
}
