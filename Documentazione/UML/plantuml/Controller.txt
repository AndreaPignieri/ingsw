@startuml
title Backend - Controller Layer & Service Dependencies

' Parametri di stile e layout
skinparam linetype ortho
skinparam classAttributeIconSize 0
skinparam BoxPadding 10
skinparam nodesep 20
skinparam ranksep 20

' Definizione Colori
skinparam package {
    BackgroundColor<<Web>> AliceBlue
    BackgroundColor<<Logic>> HoneyDew
}

package "com.dietiestates25.controller" <<Web>> {
    class AuthController {
        +register(RegisterRequest)
        +registerAgency(AgencyRegisterRequest)
        +login(LoginRequest) : AuthResponse
    }

    class PropertyController {
        +searchProperties(...)
        +createProperty(req, Principal) : PropertyDTO
        +getProperty(id) : PropertyDTO
    }

    class AgentController {
        +getAgents() : List<AgentDTO>
        +getAgent(id) : AgentDTO
        +updateAgent(id, req)
    }

    class AgencyController {
        +getAgency(id) : AgencyDTO
        +getAgentsByAgency(id) : List<AgentDTO>
    }

    class UserController {
        +getCurrentUser() : UserDTO
        +updateProfile(req)
    }

    class ReviewController {
        +getReviewsByAgent(id) : List<ReviewDTO>
        +createReview(req)
    }

    class FileUploadController {
        +uploadFile(file)
    }
}

' Infrastruttura Service Layer (solo nomi per mostrare dipendenze)
package "com.dietiestates25.service" <<Logic>> {
    class AuthService
    class PropertyService
    class AgentService
    class AgencyService
    class UserService
    class ReviewService
    interface StorageService <<interface>>
}

' Relazioni verticali per evitare l'effetto "stringa orizzontale"
controller ---> service

@enduml
