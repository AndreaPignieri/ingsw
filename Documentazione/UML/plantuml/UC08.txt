@startuml
skinparam style strictuml
skinparam SequenceMessageAlignment center
skinparam BoxPadding 10


title UC08 - Creazione annuncio immobile

actor "Agente" as A

box "Frontend" #AliceBlue
    participant "PropertyCreateComponent" as FE
    participant "FileUploadService" as FUS
end box

box "Backend (Spring Boot)" #HoneyDew
    participant "PropertyController" as PC
    participant "PropertyService" as PS
    participant "AgentRepository" as AR
    participant "AmenityRepository" as AMR
    participant "PropertyRepository" as PR
end box

box "DataBase" #BlanchedAlmond
    database "DataBase" as DB
end box

== Compilazione form e upload foto ==

A -> FE : Apre pagina "Create Property"
activate FE
FE --> A : Mostra form vuoto

A -> FE : Compila campi e seleziona immagini
FE -> FUS : uploadFile(file)
activate FUS
FUS -> DB : Salva metadati / Upload S3
DB --> FUS : OK
FUS --> FE : URL foto (es. https://...)
deactivate FUS

note right of FE : Inserisce URL foto ricevuti\nnella request finale

== Invio creazione annuncio ==

A -> FE : Clic su "Save Property"
FE -> PC : POST /properties (JWT + Body)
activate PC

PC -> PS : createProperty(request, agentEmail)
activate PS

PS -> AR : findByEmail(agentEmail)
activate AR
AR -> DB : SELECT * FROM agent
DB --> AR : Agent object
AR --> PS : Agent
deactivate AR

PS -> PS : Valida Agency & \nMappa DTO -> Entity

loop per ogni amenity
    PS -> AMR : findByName(amenity)
    AMR -> DB : SELECT/INSERT
    AMR --> PS : Amenity entity
end

PS -> PR : save(property)
activate PR
PR -> DB : INSERT property & relazioni
DB --> PR : Saved Property (ID generato)
PR --> PS : Property entity
deactivate PR

PS --> PC : PropertyDTO
deactivate PS

PC --> FE : 201 Created + PropertyDTO
deactivate PC

FE --> A : Messaggio successo
deactivate FE

== Estensioni (Gestione Errori) ==

PS -[#red]> PC : Eccezione (Validazione fallita)
activate PC
PC -[#red]> FE : 4xx / 5xx Error Response
deactivate PC
FE -[#red]> A : Mostra alert errore

@enduml
