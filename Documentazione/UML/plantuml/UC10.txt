@startuml
skinparam style strictuml
skinparam SequenceMessageAlignment center
skinparam BoxPadding 10

title UC10 - Gestione profilo agente (biografia e foto)

actor "Agente" as A

box "Frontend" #AliceBlue
    participant "ProfileComponent" as FE
    participant "FileUploadService" as FUS
end box

box "Backend" #HoneyDew
    participant "UserController" as UC
    participant "UserService" as US
    participant "UserRepository" as UR
    participant "AgentRepository" as AR
end box

box "DataBase" #BlanchedAlmond 
   database "DataBase" as DB
end box

== Caricamento profilo ==

A -> FE : Naviga su /profile
activate FE
FE -> UC : GET /users/me
activate UC
UC -> US : getCurrentUserProfile(principal)
activate US
US -> UR : findByEmail(email)
activate UR
UR -> DB : SELECT users JOIN agent
activate DB
DB --> UR : User/Agent Data
deactivate DB
UR --> US : User/Agent Entity
deactivate UR
US --> UC : AgentDTO
deactivate US
UC --> FE : JSON profilo
deactivate UC
FE --> A : Mostra dati e foto

== Upload nuova foto (opzionale) ==

A -> FE : Seleziona nuova foto profilo
FE -> FUS : uploadFile(resizedFile)
activate FUS
FUS -> DB : Storage S3 / Metadata
FUS --> FE : URL nuova foto
deactivate FUS
note right of FE : Aggiorna localmente\nil campo profilePhoto

== Salvataggio profilo ==

A -> FE : Clic su "Save Changes"
FE -> UC : PUT /users/me (Body: DTO)
activate UC
UC -> US : updateProfile(updateRequest)
activate US

US -> UR : findByEmail(email)
activate UR
UR -> DB : SELECT
DB --> UR : User/Agent
deactivate UR

US -> US : Aggiorna campi\nEntity (Bio, Photo, etc.)

group Transazione Aggiornamento
    US -> UR : save(userEntity)
    UR -> DB : UPDATE users
    US -> AR : save(agentEntity)
    AR -> DB : UPDATE agent
end

US --> UC : AgentDTO aggiornato
deactivate US
UC --> FE : 200 OK + DTO
deactivate UC
FE --> A : Messaggio "Profile updated"
deactivate FE

@enduml
