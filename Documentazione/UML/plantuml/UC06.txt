@startuml
skinparam style strictuml
skinparam SequenceMessageAlignment center
skinparam BoxPadding 10

title UC06 - Ricerca avanzata immobili

actor User as U

box "Frontend (Angular)" #AliceBlue
    participant "Home / PropertyList\nComponent" as FE
    participant "MapSearchComponent" as MAP
end box

box "Backend (Spring Boot)" #HoneyDew
    participant "PropertyController" as PC
    participant "PropertyService" as PS
    participant "PropertyRepository" as PR
end box

box "DataBase" #BlanchedAlmond
    database "DataBase" as DB
end box

U -> FE : Inserisce filtri (città, tipo, prezzi, ...)\nClic su "Search"
activate FE
FE -> PC : GET /properties?city=...&type=...
activate PC
PC -> PS : searchProperties(criteria, page, limit)
activate PS

PS -> PR : findAll(specification(criteria), pageable)
activate PR
PR -> DB : Esegue query con filtri\n(Spatial ST_Distance)
activate DB
DB --> PR : Page<Property>
deactivate DB
PR --> PS : Page<Property>
deactivate PR

PS --> PC : Page<PropertyDTO>
deactivate PS
PC --> FE : JSON Page<PropertyDTO>
deactivate PC

FE -> MAP : updateMarkers(propertiesDTO)
activate MAP
MAP --> FE : Marker aggiornati
deactivate MAP

FE --> U : Mostra lista immobili + mappa
deactivate FE

== Variante con modifica raggio/map center ==

U -> MAP : Drag cerchio / Click mappa\n(cambia centro/raggio)
activate MAP
MAP -> FE : searchAreaChange(lat, lng, radius)
deactivate MAP

activate FE
FE -> PC : GET /properties?lat=...&lng=...&radius=...
note over PC, DB : Il flusso segue la logica descritta precedentemente
FE --> U : Aggiorna risultati
deactivate FE

@enduml
