' Stile per le entità del DB

    BackgroundColor<<Entity>> BlanchedAlmond
    BorderColor Black
    BackgroundColor<<Enum>> OldLace
}

package "User Management" {
    class User <<Entity>> {
        id: Long
        email: String
        passwordHash: String
        firstName: String
        lastName: String
        registeredAt: LocalDateTime
        isActive: Boolean
        authProvider: String
    }

    class Agent <<Entity>> {
        biography: String
        profilePhoto: String
        birthDate: LocalDate
        phoneNumber: String
    }

    class Agency <<Entity>> {
        id: Long
        name: String
        email: String
        phone: String
        address: String
        createdAt: LocalDateTime
    }

    enum Role <<Enum>> {
        USER
        AGENT
        AGENCY
    }
}

package "Property Management" {
    class Property <<Entity>> {
        id: Long
        title: String
        description: String
        price: BigDecimal
        sizeSqm: Integer
        rooms: Integer
        floor: Integer
        bathrooms: Integer
        condition: String
        address: String
        city: String
        location: Point
        publishedAt: LocalDateTime
    }

    class PropertyPhoto <<Entity>> {
        id: Long
        url: String
    }

    class Amenity <<Entity>> {
        id: Long
        name: String
    }

    enum PropertyType <<Enum>> {
        SALE
        RENT
    }

    enum EnergyClass <<Enum>> {
        A1, A2, B, C, D, E, F, G
    }
}

package "Feedback" {
    class Review <<Entity>> {
        id: Long
        score: Integer
        comment: String
        createdAt: LocalDateTime
    }
}

' Gerarchia
User <|-- Agent

' Relazioni
Agency "1" o-- "0..*" User
Agency "1" o-- "0..*" Property

User "0..*   " -- "1" Role

Property "1" o-- "0..*" PropertyPhoto
Property "0..*" -- "0..*" Amenity

Agent "1" o-- "0..*" Property
Agent "1" o-- "   0..*" Review
User "1 " o-- "0..*" Review

' Relazioni Enum
Property --> PropertyType
Property --> EnergyClass

@enduml
